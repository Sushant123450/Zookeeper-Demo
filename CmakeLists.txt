cmake_minimum_required(VERSION 3.20)

project(ZKeeper LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Protobuf CONFIG REQUIRED)
find_package(gRPC CONFIG REQUIRED)

set(PROTO_FILE "${CMAKE_CURRENT_SOURCE_DIR}/proto/node.proto")
set(GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")

file(MAKE_DIRECTORY "${GENERATED_DIR}")

set(PROTO_SRCS
    "${GENERATED_DIR}/node.pb.cc"
    "${GENERATED_DIR}/node.grpc.pb.cc"
)

set(PROTO_HDRS
    "${GENERATED_DIR}/node.pb.h"
    "${GENERATED_DIR}/node.grpc.pb.h"
)

add_custom_command(
    OUTPUT ${PROTO_SRCS} ${PROTO_HDRS}
    COMMAND $<TARGET_FILE:protobuf::protoc>
    ARGS
        --proto_path "${CMAKE_CURRENT_SOURCE_DIR}/proto"
        --cpp_out "${GENERATED_DIR}"
        --grpc_out "${GENERATED_DIR}"
        --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin>
        "${PROTO_FILE}"
    DEPENDS "${PROTO_FILE}"
)

# ✅ Proto library
add_library(zkeeper_proto ${PROTO_SRCS} ${PROTO_HDRS})
target_include_directories(zkeeper_proto PUBLIC "${GENERATED_DIR}")

target_link_libraries(zkeeper_proto
    PUBLIC
    gRPC::grpc++_unsecure
    protobuf::libprotobuf
)

# ✅ Server
add_executable(Node node/server.cpp)
target_link_libraries(Node PRIVATE zkeeper_proto)

# ✅ Client
add_executable(Client node/client.cpp)
target_link_libraries(Client PRIVATE zkeeper_proto)
